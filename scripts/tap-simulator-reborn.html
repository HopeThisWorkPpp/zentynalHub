local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")

local lp = Players.LocalPlayer
local remotes = ReplicatedStorage:WaitForChild("Remotes", 10)

local cfg = {
    AutoMine    = true,
    AutoHatch   = true,
    AutoCollect = true,
    AutoCraft   = true,
    TapDelay    = 0.05,
    SafeMode    = true,
}

local function safeCall(fn, ...)
    local ok, err = pcall(fn, ...)
    if not ok then warn("[ZH] " .. tostring(err)) end
end

local function findRemote(name)
    return remotes and remotes:FindFirstChild(name)
end

local tapRemote = findRemote("Tap") or findRemote("Mine") or findRemote("TapMine")

task.spawn(function()
    while cfg.AutoMine do
        local d = cfg.SafeMode and (cfg.TapDelay + math.random(1,3)*0.01) or cfg.TapDelay
        task.wait(d)
        safeCall(function()
            if tapRemote then tapRemote:FireServer() end
        end)
    end
end)

task.spawn(function()
    while cfg.AutoHatch do
        task.wait(1)
        safeCall(function()
            local r = findRemote("Hatch") or findRemote("HatchEgg")
            if r then
                local eggs = workspace:FindFirstChild("Eggs")
                if eggs then
                    local egg = eggs:FindFirstChildOfClass("Model")
                    if egg then r:FireServer(egg) end
                end
            end
        end)
    end
end)

task.spawn(function()
    while cfg.AutoCollect do
        task.wait(0.1)
        safeCall(function()
            local r = findRemote("Collect") or findRemote("PickupCoin")
            local items = workspace:FindFirstChild("Collectibles") or workspace:FindFirstChild("Drops")
            if items and r then
                for _, obj in ipairs(items:GetChildren()) do r:FireServer(obj) end
            end
        end)
    end
end)

task.spawn(function()
    while cfg.AutoCraft do
        task.wait(2)
        safeCall(function()
            local r = findRemote("Craft") or findRemote("CraftItem")
            if r then r:FireServer() end
        end)
    end
end)
